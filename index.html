function buildShuffledOrder(){

  const flat = [];

  GIFTS.forEach(g => g.statements.forEach((s, si) => flat.push({ giftId: g.id, si, text: s })));

  // Fisher–Yates shuffle so statements appear in random order, never grouped by gift

  for(let i = flat.length - 1; i > 0; i--){

    const j = Math.floor(Math.random() * (i + 1));

    [flat[i], flat[j]] = [flat[j], flat[i]];

  }

  return flat;

}

}


/* ---------------- RENDER: QUIZ ---------------- */

function renderQuiz(){

  const total = state.order.length;

  const totalPages = Math.ceil(total / BATCH_SIZE);

  const start = state.page * BATCH_SIZE;

  const batch = state.order.slice(start, start + BATCH_SIZE);

  const pct = Math.round((start / total) * 100);


  const statementsHtml = batch.map((item, idx) => {

    const qNum = start + idx + 1;

    const answered = state.answers[item.giftId][item.si];

    return `

    <fieldset>

      <div class="statement">${qNum}. ${item.text}</div>

      <div class="scale" data-gift="${item.giftId}" data-si="${item.si}">

        ${SCALE.map(opt => `

          <label class="${answered===opt.v ? 'checked':''}" data-val="${opt.v}">

            <input type="radio" name="q-${item.giftId}-${item.si}" value="${opt.v}" ${answered===opt.v?'checked':''}>

            <span class="num">${opt.v}</span>${opt.label}

          </label>

        `).join('')}

      </div>

    </fieldset>

  `;

  }).join('');


  app.innerHTML = `

    <div class="progress-wrap">

      <div class="progress-track"><div class="progress-fill" style="width:${pct}%"></div></div>

      <div class="progress-label"><span>Question ${start+1}–${Math.min(start+BATCH_SIZE, total)} of ${total}</span><span>${pct}% complete</span></div>

    </div>

    <div class="card">

      <div class="gift-name" style="font-size:18px;">Rate each statement</div>

      <p class="gift-def">How true is each of the following of you? Statements are shown in random order and aren't grouped by topic — answer based on your first instinct.</p>

      ${statementsHtml}

      <div class="btn-row">

        <button class="btn btn-ghost" id="prevBtn" ${state.page===0?'disabled':''}>← Back</button>

        <button class="btn btn-primary" id="nextBtn">${state.page===totalPages-1 ? 'See my results →' : 'Next →'}</button>

      </div>

    </div>

  `;


  // wire up scale click handlers

  app.querySelectorAll('.scale').forEach(scaleEl => {

    const giftId = scaleEl.dataset.gift;

    const si = parseInt(scaleEl.dataset.si, 10);

    scaleEl.querySelectorAll('label').forEach(lbl => {

      lbl.addEventListener('click', () => {

        const val = parseInt(lbl.dataset.val, 10);

        state.answers[giftId][si] = val;

        scaleEl.querySelectorAll('label').forEach(l => l.classList.remove('checked'));

        lbl.classList.add('checked');

      });

    });

  });


  document.getElementById('prevBtn').onclick = () => {

    if(state.page>0){ state.page--; render(); window.scrollTo({top:0,behavior:'smooth'}); }

  };

  document.getElementById('nextBtn').onclick = () => {

    const incomplete = batch.some(item => state.answers[item.giftId][item.si] === null);

    if(incomplete){

      alert("Please rate every statement before continuing.");

      return;

    }

    if(state.page < totalPages-1){

      state.page++;

      render();

    } else {

      state.stage = "results";

      render();

    }

    window.scrollTo({top:0,behavior:'smooth'});

  };

}


/* ---------------- SCORING ---------------- */

function computeScores(){

  return GIFTS.map(g => {

    const sum = state.answers[g.id].reduce((a,b)=>a+b,0);

    return { ...g, score: sum, pct: Math.round((sum/16)*100) };

  }).sort((a,b) => b.score - a.score);

}

/* ---------------- CSV EXPORT (local fallback) ---------------- */

function downloadCSV(ranked){

  const rows = [

    ["Timestamp","First Name","Last Name","Email","Gift","Score (out of 16)","Percent","Rank"],

    ...ranked.map((g,i) => [

      new Date().toISOString(), state.profile.firstName, state.profile.lastName, state.profile.email,

      g.name, g.score, g.pct+"%", i+1

    ])

  ];

  const csv = rows.map(r => r.map(v => `"${String(v).replace(/"/g,'""')}"`).join(',')).join('\n');

  const blob = new Blob([csv], {type:'text/csv'});

  const a = document.createElement('a');

  a.href = URL.createObjectURL(blob);

  a.download = `spiritual-gifts-${state.profile.firstName}-${state.profile.lastName}.csv`.replace(/\s+/g,'-');

  a.click();

}


/* ---------------- ZAPIER WEBHOOK EXPORT ----------------

   Every respondent's results are automatically POSTed as JSON to a fixed

   Zapier "Catch Hook" webhook the moment the results page loads — no setup

   step or user action required. From there the Zap routes it into Google

   Sheets, Airtable, a CRM, email, Slack, etc. */

const ZAPIER_WEBHOOK_URL = "https://hooks.zapier.com/hooks/catch/26703624/46c3a9p/";


function buildZapierPayload(ranked){

  const top5 = ranked.slice(0,5);

  const payload = {

    timestamp: new Date().toISOString(),

    first_name: state.profile.firstName,

    last_name: state.profile.lastName,

    email: state.profile.email,

  };

  top5.forEach((g,i) => {

    payload[`top_gift_${i+1}`] = g.name;

    payload[`top_gift_${i+1}_score`] = g.score;

  });

  payload.all_scores_json = JSON.stringify(ranked.map(g => ({ gift: g.name, score: g.score })));

  return payload;

}


async function sendToZapier(ranked){

  const statusEl = () => document.getElementById('exportStatus');

  if(statusEl()) statusEl().innerHTML = `<div class="export-status pending"><span class="spinner"></span>Saving your results…</div>`;

  try{

    const res = await fetch(ZAPIER_WEBHOOK_URL, {

      method: "POST",

      mode: "cors",

      headers: { "Content-Type": "application/json" },

      body: JSON.stringify(buildZapierPayload(ranked))

    });

    if(!res.ok){

      throw new Error(`Webhook responded with status ${res.status}`);

    }

    if(statusEl()) statusEl().innerHTML = `<div class="export-status ok">✓ Your results have been saved.</div>`;

  } catch(err){

    if(statusEl()) statusEl().innerHTML = `<div class="export-status err">Couldn't save your results automatically (${escapeHtml(err.message)}). Please use "Download CSV" below and send it to us directly.</div>`;

  }

}


// Auto-send exactly once per results view (guards against re-render loops).

let zapierSentThisSession = false;

function autoSendToZapier(ranked){

  if(zapierSentThisSession) return;

  zapierSentThisSession = true;

  sendToZapier(ranked);

}


function escapeHtml(str){

  const d = document.createElement('div');

  d.textContent = str;

  return d.innerHTML;

}

  else if(state.stage === "quiz") renderQuiz();

  else renderResults();

}
